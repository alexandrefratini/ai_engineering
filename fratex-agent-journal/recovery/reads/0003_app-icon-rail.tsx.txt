101	function isActive(item: RailItem, pathname: string): boolean {
102		return (
103			pathname === item.href ||
104			(item.match === "prefix" && pathname.startsWith(item.href)) ||
105			Boolean(item.related?.some((prefix) => pathname.startsWith(prefix)))
106		);
107	}

109	function RailLink({
110		item,
111		active,
112		onPrefetch,
113	}: {
114		item: RailItem;
115		active: boolean;
116		onPrefetch: () => void;
117	}) {
118		return (
119			<Tooltip>
120				<TooltipTrigger asChild>
121					<Button
122						asChild
123						variant="ghost"
124						size="icon"
125						className={cn(
126							"text-muted-foreground",
127							active &&
128								"bg-muted text-foreground hover:bg-muted hover:text-foreground",
129						)}
130					>
131						<Link
132							href={item.href}
133							prefetch
134							onMouseEnter={onPrefetch}
135							onFocus={onPrefetch}
136							aria-current={active ? "page" : undefined}
137							transitionTypes={["nav-lateral"]}
138						>
139							<Icon icon={item.icon} className={item.iconClassName} />
140							<span className="sr-only">{item.title}</span>
141						</Link>
142					</Button>
143				</TooltipTrigger>
144				<TooltipContent side="right">{item.title}</TooltipContent>
145			</Tooltip>
146		);
147	}

248	export function AppIconRail() {
249		const pathname = usePathname();
250		const workspaceUrl = useWorkspaceUrl();
251		const { open, setOpen } = useMobileNav();
252		const prefetchSection = usePrefetchSection();
253	
254		const items = useMemo(
255			() =>
256				ITEMS.map((item) => ({
257					...item,
258					section: item.href,
259					href: workspaceUrl(item.href),
260					related: item.related?.map((path) => workspaceUrl(path)),
261				})),
262			[workspaceUrl],
263		);
264		const inChat = items.some(
265			(item) => item.title === "Chat" && isActive(item, pathname),
266		);
267	
268		return (
269			<>
270				<nav
271					aria-label="Primary"
272					className="hidden w-14 shrink-0 flex-col items-center gap-1 border-r py-3 md:flex [view-transition-name:app-rail]"
273				>
274					{items.map((item) => (
275						<RailLink
276							key={item.href}
277							item={item}
278							active={isActive(item, pathname)}
279							onPrefetch={() => prefetchSection(item.section)}
280						/>
281					))}
282				</nav>
283	
284				<Sheet open={open} onOpenChange={setOpen}>
285					{inChat ? (
286						<SheetContent
287							side="left"
288							showCloseButton={false}
289							className="w-5/6 max-w-sm flex-row gap-0 p-0"
290						>
291							<SheetHeader className="sr-only">
292								<SheetTitle>Navigation and agent chats</SheetTitle>
293							</SheetHeader>
294							<nav
295								aria-label="Primary"
296								className="flex w-14 shrink-0 flex-col items-center gap-1 border-r py-3"
297							>
298								<Button
299									variant="ghost"
300									size="icon"
301									aria-label="Close navigation"
302									onClick={() => setOpen(false)}
303								>
304									<Icon icon={Close} />
305								</Button>
306								<div className="my-1 h-px w-5 bg-border" />
307								{items.map((item) => (
308									<MobileRailIconLink
309										key={item.href}
310										item={item}
311										active={isActive(item, pathname)}
312										onNavigate={() => setOpen(false)}
313										onPrefetch={() => prefetchSection(item.section)}
314									/>
315								))}
316							</nav>
317							<AgentBuilderSidebar
318								className="flex flex-1"
319								onNavigate={() => setOpen(false)}
320							/>
321						</SheetContent>
322					) : (
323						<SheetContent side="left" className="w-64 gap-0 p-0">
324							<SheetHeader>
325								<SheetTitle>Navigation</SheetTitle>
326							</SheetHeader>
327							<nav
328								aria-label="Primary"
329								className="flex flex-1 flex-col gap-1 p-2"
330							>
331								{items.map((item) => (
332									<MobileRailLink
333										key={item.href}
334										item={item}
335										active={isActive(item, pathname)}
336										onNavigate={() => setOpen(false)}
337										onPrefetch={() => prefetchSection(item.section)}
338									/>
339								))}
340							</nav>
341						</SheetContent>
342					)}
343				</Sheet>
344			</>
345		);
346	}
